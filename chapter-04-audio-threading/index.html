🐍 ໂຄ້ດສ້າງສຽງ "Beep, beep, beep!" (`audio_manager.py`)

ດັງ 3 ຈັງຫວະຕິດຕໍ່ກັນຜ່ານ Background Thread

def _sound_worker(self):
    # ສຳລັບ Windows: ດັງ Beep, beep, beep! (1200Hz 3 ບາດ)
    if platform.system() == "Windows":
        import winsound
        for _ in range(3):
            winsound.Beep(1200, 120)
            time.sleep(0.08)
        return

    # ສຳລັບ WSL (Linux on Windows):
    ps_cmd = (
        "[console]::beep(1200,120); Start-Sleep -m 70; "
        "[console]::beep(1200,120); Start-Sleep -m 70; "
        "[console]::beep(1200,120)"
    )
    subprocess.run(
        ["powershell.exe", "-NoProfile", "-Command", ps_cmd],
        check=False
    )
