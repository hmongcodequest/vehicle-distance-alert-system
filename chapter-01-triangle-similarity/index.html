# =====================================================================
# ບົດທີ 1: ការຂຽນສູດສາມແຈຄ້າຍຄືໃນພາສາ Python (triangle_math.py)
# =====================================================================

KNOWN_WIDTH = 1.8    # W: ຄວາມກວ້າງຕົວຈິງຂອງລົດເກັງ (ແມັດ)
FOCAL_LENGTH = 800.0 # F: ຄ່າຄວາມຍາວໂຟກັສຂອງກ້ອງ (Pixels)
SAFE_DISTANCE = 30.0 # ເກນແຈ້ງເຕືອນໄລຍະຫ່າງ (30 ແມັດ)

def calculate_distance(pixel_width: float) -> float:
    """ສູດສາມແຈຄ້າຍຄື: D = (W * F) / P"""
    return (KNOWN_WIDTH * FOCAL_LENGTH) / pixel_width

# ທົດສອບດ້ວຍຄ່າ Pixel ທີ່ເຈົ້າກຳລັງເລືອກຢູ່ເທິງແຜນວາດ:
pixel_p = 48.0
distance_d = calculate_distance(pixel_p)

print(f"ຄວາມກວ້າງໃນພາບ P = {pixel_p:.1f} px -> ໄລຍະຫ່າງ D = {distance_d:.1f} ແມັດ")
if distance_d < SAFE_DISTANCE:
    print("⚠️ WARNING: ລົດຄັນໜ້າຢູ່ໃກ້ກວ່າ 30 ແມັດ!")
else:
    print("✅ SAFE: ໄລຍະຫ່າງປອດໄພ (>= 30 ແມັດ)")

🧮 ການຄິດໄລ່ສູດຕົວຈິງ (Live Math)

ສູດ: D = (W × F) / P

1.8m × 800px 48.0 px

W
Known Width (1.8 ແມັດ) ຄວາມກວ້າງຕົວຈິງໂດຍສະເລ່ຍຂອງລົດເກັງ
F
Focal Length (800) ຄ່າຄວາມຍາວໂຟກັສຂອງກ້ອງທີ່ຜ່ານການ Calibrate ແລ້ວ
P
Pixel Width (48.0 px) ຄວາມກວ້າງຂອງກ່ອງລົດໃນພາບ (x2 - x1)

💡 ວິທີຫາຄ່າ F ໃນ Python (Calibration)

ກ່ອນຈະເອົາໂຄ້ດໄປໃຊ້ຈິງ ເຮົາຕ້ອງຫາຄ່າ F ຂອງກ້ອງກ່ອນ ໂດຍການຈອດລົດຫ່າງອອກໄປ 30 ແມັດພໍດີ ແລ້ວອ່ານຄ່າ Pixel (ສົມມຸດໄດ້ 48 px): 

# ສູດປີ້ນກັບເພື່ອຫາຄ່າ Focal Length (F)
P = 48.0  # ຄວາມກວ້າງກ່ອງລົດໃນພາບ (Pixels)
D = 30.0  # ໄລຍະຈອດລົດຕົວຈິງ (ແມັດ)
W = 1.8   # ຄວາມກວ້າງລົດຕົວຈິງ (ແມັດ)

FOCAL_LENGTH = (P * D) / W
print(FOCAL_LENGTH)  # ຜົນລັບ: 800.0
